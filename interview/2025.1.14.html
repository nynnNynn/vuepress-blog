// 用 JS 实现二分查找（折半查找）算法，并写出时间复杂度
// 要求：若target在arr中返回true，否则为false
// function find(arr, target) {
// // TODO
// }
// arr为数字不重复的升序数组，如 [1,5,9]
// target为目标数字
function test(arr, target) {
  function search(arr, target, accu) {
    let pivot = arr[arr.length / 2];

    if (accu > 0) {
      if (target < arr[0]) {
        return false;
      }
    }

    if (accu === 0) {
      if (target > arr[arr.length - 1]) {
        return false;
      }
    }
    if (target === pivot) {
      return pivot + accu;
    }
    if (target > pivot) {
      let temp = arr.slice(arr.length / 2, arr.length - 1);
      return search(temp, target, arr.length / 2);
    } else {
      let temp = arr.slice(0, arr.length / 2);
      return search(temp, target, 0);
    }
  }
  return search(arr, target, 0);
}
