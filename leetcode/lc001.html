# lc01两数和

# 代码

/**
 * @param {number[]} nums
 * @param {number} target
 * @return {number[]}
 */
var twoSum = function (nums, target) {
  let result = {};
  for (let i = 0; i < nums.length; i++) {
    if (result[target - nums[i]] || result[target - nums[i]] === 0) {
      return [result[target - nums[i]], i];
    }

    result[nums[i]] = i;
  }
};
