# lc056合并区间

# 代码

intervals.sort(
    (a,b) => {
      return a[0] - b[0];
    }
  )
  return intervals.reduce((accu, curr) => {
    if(accu.length < 1){
      accu.push(curr);
      return accu;
    }else{
      if(accu[accu.length - 1][1] >= curr[0]){
        if(curr[1] <= accu[accu.length - 1][1]){
          return accu;
        }
        //[1,5] [2,6]/[5,6]
        accu[accu.length - 1] = [accu[accu.length - 1][0], curr[1]];
        return accu;
        
      }else{
        accu.push(curr);
        return accu;
      }
    }

  }, [])

# 思路

  • 没有更好的办法，排序后遍历操纵最后一个就是目前来看唯一且比较靠谱的解法
  • 可以不用reduce，区别就是指针去指

# 备注

  • 就是关于数组传参，做引用，包括那个二维数组那种情况，这个一直是一个我会混淆的地方，我怀疑能追溯到c+时期，其实这个事情要注意一下的， 主要就是空数组时候的直接赋值变push这个问题
  • 注意边界条件的几重判断，注意原题是没sort的
